test('Play/Pause/Stop buttons change transport state', async ({ page }) => {
  const errors: string[] = []
  page.on('pageerror', err => errors.push(err.message))

  await page.goto('examples/transport-sequencer')
  await page.waitForSelector('.transport-buttons', { timeout: 10000 })
  await page.waitForLoadState('networkidle')

  // Play button visible, Pause button hidden initially
  expect(await page.locator('.play-btn').isVisible()).toBe(true)
  expect(await page.locator('.pause-btn').isVisible()).toBe(false)

  // Click Play — pause button should appear within 30s (piano.js is 1.4MB)
  await page.locator('.play-btn').click()
  await page.waitForFunction(
    () => document.querySelector('.pause-btn') !== null && (document.querySelector('.pause-btn') as HTMLElement).style.display !== 'none',
    { timeout: 30000 }
  )

  // BPM slider is present
  const bpmSlider = page.locator('[aria-label^="BPM:"]')
  expect(await bpmSlider.isVisible()).toBe(true)

  // Stop button works

  await page.waitForFunction(
    () => document.querySelector('.play-btn') !== null && (document.querySelector('.play-btn') as HTMLElement).style.display !== 'none',

**Test 4 (TSEQ-04): Playhead advancement after Play**
```typescript
test('Playhead advances after transport starts', async ({ page }) => {
  const errors: string[] = []
  page.on('pageerror', err => errors.push(err.message))

  await page.goto('examples/transport-sequencer')
  await page.waitForSelector('.step-grid', { timeout: 10000 })
  await page.waitForLoadState('networkidle')

  // Get initial playhead count
  const getPlayheadCount = () =>
    document.querySelectorAll('.step-cell.playhead').length

  const initialCount = await page.evaluate(getPlayheadCount)
  expect(initialCount).toBeGreaterThan(0)  // At least one playhead cell should exist (step 0 highlighted)

  // Start transport
  await page.locator('.play-btn').click()
  await page.waitForFunction(
    () => document.querySelector('.pause-btn') !== null,
    { timeout: 30000 }
  )

  // Wait for playhead to advance to a different step
  const initialStep = await page.evaluate(() => {
    const cells = document.querySelectorAll('.step-cell')
    for (let i = 0; i < cells.length; i++) {
      if (cells[i].classList.contains('playhead')) return i
    }
    return -1
  })

  await page.waitForFunction(
    (step) => {
      const cells = document.querySelectorAll('.step-cell')
      for (let i = 0; i < cells.length; i++) {
        if (cells[i].classList.contains('playhead') && i !== step) return true
      }
      return false
    },
    initialStep,
    { timeout: 600 }
  )

  expect(errors, 'playhead advancement should have no page errors').toHaveLength(0)
})
```
