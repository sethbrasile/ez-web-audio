```typescript
const createdSources: any[] = []
vi.spyOn(audioContext, 'createBufferSource').mockImplementation(() => {
  const source = {
    buffer: null,
    loop: false,
    connect: vi.fn().mockReturnThis(),
    disconnect: vi.fn(),
    start: vi.fn(),
    stop: vi.fn(),
    onended: null,
  } as unknown as AudioBufferSourceNode
  createdSources.push(source)
  return source
})
```

Note: `stop(name)` only works on looping sprites — non-looping sprites are tracked by `activeSources` only when `loop: true` in the sprite definition (check sprite.ts line 189 comment). For non-looping sprites, stop() is a no-op because they're not in activeSources.
