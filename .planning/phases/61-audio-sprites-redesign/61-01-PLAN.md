Add Howler-style sprite manifest support to the library. Format detection is automatic.

Purpose: Developers coming from Howler.js can paste their existing manifests directly into createSprite() without manual conversion.

       ```typescript
       export function normalizeManifest(manifest: SpriteManifest): AudiospriteManifest {
         if (!isHowlerManifest(manifest)) {
           return manifest
         }
         const spritemap: Record<string, SpriteDefinition> = {}
         for (const [name, tuple] of Object.entries(manifest.sprite)) {
           spritemap[name] = {
             start: tuple[0] / 1000,
             end: (tuple[0] + tuple[1]) / 1000,
             loop: tuple[2] ?? false,
           }
         }
         return { spritemap }
       }
       ```
