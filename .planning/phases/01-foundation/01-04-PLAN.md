       // Emit resume event before starting
       this.emit('resume', {

      sound.stop();

      const stopHandler = vi.fn();
      sound.on('play', playHandler).on('stop', stopHandler);
      sound.play();
      sound.stop();
      expect(playHandler).toHaveBeenCalledTimes(1);
      expect(stopHandler).toHaveBeenCalledTimes(1);
    });
  });

  describe('.once()', () => {
    it('fires handler only once', () => {
      const handler = vi.fn();
      sound.once('play', handler);
      sound.play();
      sound.stop();
      sound.play();
      expect(handler).toHaveBeenCalledTimes(1);
    });
  });

  describe('.off()', () => {
    it('removes specific handler', () => {
      const handler = vi.fn();
      sound.on('play', handler);
      sound.off('play', handler);
      sound.play();
      expect(handler).not.toHaveBeenCalled();
    });
  });
