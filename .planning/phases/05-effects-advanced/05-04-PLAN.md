1. **src/debug/messages.ts** - Message types and formatting:
```typescript
export interface DebugMessage {
  type: 'event' | 'connection' | 'warning'
  source: string      // Sound name or identifier
  message: string     // Human-readable message
  timestamp: number   // audioContext.currentTime or Date.now()
  details?: Record<string, unknown>
}

export function formatDebugMessage(msg: DebugMessage): string {
  const prefix = `[ez-audio:${msg.type}]`
  const time = msg.timestamp.toFixed(3)
  return `${prefix} [${time}] ${msg.source}: ${msg.message}`
}
```

2. **src/debug/logger.ts** - Logger state and handler:
```typescript
let globalDebugEnabled = false
let debugHandler: ((msg: DebugMessage) => void) | null = null

export function setGlobalDebug(enabled: boolean): void {
  globalDebugEnabled = enabled
}

export function isGlobalDebugEnabled(): boolean {
  return globalDebugEnabled
}

export function setHandler(handler: ((msg: DebugMessage) => void) | null): void {
  debugHandler = handler
}

export function getHandler(): ((msg: DebugMessage) => void) | null {
  return debugHandler
}

function defaultHandler(msg: DebugMessage): void {
  console.log(formatDebugMessage(msg), msg.details ?? '')
}

export function log(msg: DebugMessage): void {
  const handler = debugHandler ?? defaultHandler
  handler(msg)
}
```

3. **src/debug/index.ts** - Public API:
```typescript
import { setGlobalDebug, setHandler, isGlobalDebugEnabled, log } from './logger'
import type { DebugMessage } from './messages'

export type { DebugMessage }
export { formatDebugMessage } from './messages'

export function setDebugMode(enabled: boolean): void {
  setGlobalDebug(enabled)
}

export function setDebugHandler(handler: ((msg: DebugMessage) => void) | null): void {
  setHandler(handler)
}

// Main logging function - short-circuits immediately when debug disabled
export function debugLog(
  source: { name?: string; debug?: boolean },
  message: Omit<DebugMessage, 'source'> & { source?: string }
): void {
  // Fast path: if neither global nor per-sound debug, return immediately
  if (!isGlobalDebugEnabled() && !source.debug) return

  // Per-sound override: explicitly false disables even when global is on
  if (source.debug === false) return

  log({
    ...message,
    source: message.source ?? source.name ?? 'unknown'
  })
}

// Convenience functions for common message types
export function debugEvent(
  source: { name?: string; debug?: boolean },
  event: string,
  timestamp: number,
  details?: Record<string, unknown>
): void {
  debugLog(source, {
    type: 'event',
    message: event,
    timestamp,
    details
  })
}

export function debugConnection(
  source: { name?: string; debug?: boolean },
  message: string,
  timestamp: number,
  details?: Record<string, unknown>
): void {
  debugLog(source, {
    type: 'connection',
    message,
    timestamp,
    details
  })
}

export function debugWarning(
  source: { name?: string; debug?: boolean },
  message: string,
  timestamp: number,
  details?: Record<string, unknown>
): void {
  debugLog(source, {
    type: 'warning',
    message,
    timestamp,
    details
  })
}
```
