```typescript
function handleKeyDown(e: KeyboardEvent) {
  const step = e.shiftKey ? 20 : 5 // Larger steps with Shift held
  const canvas = canvasRef.value // or whatever the canvas ref is named
  if (!canvas) return

  const logicalWidth = Number(canvas.dataset.logicalWidth || canvas.width)
  const logicalHeight = Number(canvas.dataset.logicalHeight || canvas.height)

  let { x, y } = currentPosition // or however position is tracked

  switch (e.key) {
    case 'ArrowRight':
      x = Math.min(logicalWidth, x + step)
      break
    case 'ArrowLeft':
      x = Math.max(0, x - step)
      break
    case 'ArrowUp':
      y = Math.max(0, y - step) // Up = higher pitch/gain
      break
    case 'ArrowDown':
      y = Math.min(logicalHeight, y + step)
      break
    default:
      return // Don't prevent default for non-arrow keys
  }

  e.preventDefault() // Prevent page scroll

  // Start oscillator if not already playing
  if (!isPlaying.value) {
    startOscillator() // or whatever the start function is
  }

  // Update position and sound
  updateFromPosition(x, y)
}
```
