```typescript
public syncTo(transport: Transport, opts: { noteType: number }): void {
  if (this._syncedTo === transport) return // already synced to this transport
  if (this._syncedTo) this.unsync() // detach from previous
  this._syncedTo = transport
  this._syncNoteType = opts.noteType
  transport._addTrack(this)
}
```

**unsync method:**
```typescript
public unsync(): void {
  if (!this._syncedTo) return
  this._syncedTo._removeTrack(this)
  this._syncedTo = null
  // Stop clears beat state, resets index — safe since we're no longer synced
  this.internalStop() // private version of stop that doesn't throw when synced
}
```

```typescript
_addTrack(track: BeatTrack): void {
  if (this._syncedTracks.has(track)) return
  this._syncedTracks.add(track)

  if (this.playing) {
    // Hot add: calculate next beat boundary for this track's noteType
    const beatDuration = (240 * track._syncNoteType) / this.bpm
    const elapsed = this.audioContext.currentTime - this.startTime
    const nextBeatOffset = beatDuration - (elapsed % beatDuration)
    const nextBeatTime = this.audioContext.currentTime + nextBeatOffset

      track,
      nextBeatTime,
      currentBeatIndex: 0,
    })
  }
}
```

**Modified schedulerTick to include track scheduling:**
```typescript
private schedulerTick(): void {
  const currentTime = this.audioContext.currentTime

  // Schedule Transport's own ticks (position tracking)
  while (this.nextTickTime < currentTime + this.scheduleAheadTime) {
    this.emitTickAt(this.nextTickTime)
    this.advancePosition()
  }

  // Schedule each synced track's beats within lookahead window
  for (const state of this.trackStates.values()) {
    const beatDuration = (240 * state.track._syncNoteType) / this.bpm
    while (state.nextBeatTime < currentTime + this.scheduleAheadTime) {
      state.track._scheduleBeatFromTransport(state.currentBeatIndex, state.nextBeatTime)
      state.nextBeatTime += beatDuration
      state.currentBeatIndex = (state.currentBeatIndex + 1) % state.track.beats.length
    }
  }
}
```
