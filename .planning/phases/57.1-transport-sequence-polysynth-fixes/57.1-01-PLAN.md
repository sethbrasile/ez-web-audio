    entry.onStop = null

    entry.onEnd = null

    if (entry.state === 'active') {
      entry.releasedAt = this.audioContext.currentTime
      entry.state = 'released'
      this._activeCount--
    }
  }
  const onEnd = (): void => {
    if (entry.state === 'released') {
      entry.state = 'available'
    }
    // Clean up references
    entry.onStop = null
    entry.onEnd = null
  }

  entry.onStop = onStop
  entry.onEnd = onEnd
  entry.oscillator.once('stop', onStop)
  entry.oscillator.once('end', onEnd)
}
